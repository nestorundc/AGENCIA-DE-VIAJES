Script SQL para Supabase 

-- ============================================================
-- SCRIPT SQL SUPABASE - AGENCIA DE VIAJES (TABLAS Y RELACIONES)
-- ============================================================

-- 1. LIMPIEZA DE TABLAS PREVIAS
DROP TABLE IF EXISTS "RESERVA" CASCADE;
DROP TABLE IF EXISTS "PAQUETE_TURISTICO" CASCADE;
DROP TABLE IF EXISTS "DESTINO" CASCADE;
DROP TABLE IF EXISTS "CLIENTE" CASCADE;

-- 2. TABLA CLIENTE
CREATE TABLE "CLIENTE" (
    "ID_Cliente" BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "Nombre" VARCHAR(50) NOT NULL,
    "Apellido" VARCHAR(50) NOT NULL,
    "DNI" VARCHAR(8) UNIQUE NOT NULL,
    "Correo_Electronico" VARCHAR(100) UNIQUE NOT NULL,
    "Telefono" VARCHAR(15)
);

-- 3. TABLA DESTINO
CREATE TABLE "DESTINO" (
    "ID_Destino" BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "Pais" VARCHAR(50) NOT NULL,
    "Ciudad" VARCHAR(50) NOT NULL,
    "Descripcion" VARCHAR(200)
);

-- 4. TABLA PAQUETE_TURISTICO
CREATE TABLE "PAQUETE_TURISTICO" (
    "ID_Paquete" BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "ID_Destino" BIGINT NOT NULL,
    "Nombre_Paquete" VARCHAR(100) NOT NULL,
    "Precio" NUMERIC(8,2) NOT NULL,
    "Dias" INTEGER NOT NULL,

    CONSTRAINT FK_PAQUETE_DESTINO 
        FOREIGN KEY ("ID_Destino") 
        REFERENCES "DESTINO"("ID_Destino") 
        ON DELETE CASCADE
);

-- 5. TABLA RESERVA
CREATE TABLE "RESERVA" (
    "ID_Reserva" BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "ID_Cliente" BIGINT NOT NULL,
    "ID_Paquete" BIGINT NOT NULL,
    "Fecha_Reserva" DATE NOT NULL DEFAULT CURRENT_DATE,
    "Estado" VARCHAR(20) NOT NULL,

    CONSTRAINT FK_RESERVA_CLIENTE 
        FOREIGN KEY ("ID_Cliente") 
        REFERENCES "CLIENTE"("ID_Cliente") 
        ON DELETE CASCADE,

    CONSTRAINT FK_RESERVA_PAQUETE 
        FOREIGN KEY ("ID_Paquete") 
        REFERENCES "PAQUETE_TURISTICO"("ID_Paquete") 
        ON DELETE CASCADE
);
